@app.get("/api/vendors/tourist") def get_tourist_vendors(): conn = get_db_connection() cur = conn.cursor() cur.execute("SELECT id,name,category,location,description,price_label,url,type,image FROM vendors WHERE type = 'safari' ORDER BY created_at DESC LIMIT 40") rows = cur.fetchall() cur.close(); conn.close() return [dict(zip(['id','name','category','location','description','price_label','url','type','image'], r)) for r in rows]
